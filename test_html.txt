        // ===== File Panel Toggle =====
        function toggleFilePanel() {
            const panel = document.getElementById('file-panel');
            const btn = document.getElementById('rail-files');
            panel.classList.toggle('collapsed');
            btn.classList.toggle('active');
        }

        // ===== VS Code Collapsible Folders =====
        function toggleFolder(labelEl) {
            const folderId = labelEl.getAttribute('data-folder-id');
            const children = document.querySelector(`[data-folder-children="${folderId}"]`);
            
            if (labelEl.classList.contains('expanded')) {
                labelEl.classList.remove('expanded');
                labelEl.classList.add('collapsed');
                children.classList.add('collapsed');
                saveFolderState(folderId, false);
            } else {
                labelEl.classList.remove('collapsed');
                labelEl.classList.add('expanded');
                children.classList.remove('collapsed');
                saveFolderState(folderId, true);
            }
        }

        function collapseAllFolders() {
            document.querySelectorAll('.panel-folder-label').forEach(label => {
                label.classList.remove('expanded');
                label.classList.add('collapsed');
            });
            document.querySelectorAll('.folder-children').forEach(c => {
                c.classList.add('collapsed');
            });
            localStorage.setItem('neo_folders', '{}');
        }

        function saveFolderState(folderId, isOpen) {
            let states = JSON.parse(localStorage.getItem('neo_folders') || '{}');
            states[folderId] = isOpen;
            localStorage.setItem('neo_folders', JSON.stringify(states));
        }

        // Restore folder states on load
        (function restoreFolderStates() {
            let states = JSON.parse(localStorage.getItem('neo_folders') || '{}');
            document.querySelectorAll('.panel-folder-label').forEach(label => {
                const folderId = label.getAttribute('data-folder-id');
                if (states[folderId] === false) {
                    label.classList.remove('expanded');
                    label.classList.add('collapsed');
                    const children = document.querySelector(`[data-folder-children="${folderId}"]`);
                    if (children) children.classList.add('collapsed');
                }
            });
        })();

        // ===== Folder Modal =====
        function openFolderModal() {
            document.getElementById('folder-modal').classList.add('show');
            setTimeout(() => document.getElementById('folder-name-input').focus(), 100);
        }
        function closeFolderModal() {
            document.getElementById('folder-modal').classList.remove('show');
        }

        // Command Palette Removed

        // Auto-fade flash messages
        setTimeout(() => {
            document.querySelectorAll('.alert').forEach(alert => {
                alert.style.transition = "opacity 0.5s ease";
                alert.style.opacity = "0";
                setTimeout(() => alert.remove(), 500);
            });
        }, 4000);

            searchInput.addEventListener('input', function() {
                const term = this.value.toLowerCase();
                document.querySelectorAll('.panel-note-item').forEach(item => {

                        item.style.display = 'block';
                    } else {
                        item.style.display = 'none';
                    }
                });
            });
        }

        document.addEventListener('keydown', function(e) {
            if (e.ctrlKey && e.key === 'k') {
                e.preventDefault();
                if (searchInput) searchInput.focus();
            }
            if (e.key === 'Escape') {
                closeFolderModal();
                if (searchInput) searchInput.blur();
            }
        });

        // Status Bar Logic
        const sbContentArea = document.getElementById('note-content');
        if (sbContentArea) {
            function updateStatusBar() {
                const text = sbContentArea.value;
                const chars = text.length;
                const words = text.trim() === '' ? 0 : text.trim().split(/\s+/).length;
                document.getElementById('sb-words').innerText = `Words: ${words}`;
                document.getElementById('sb-chars').innerText = `Chars: ${chars}`;
            }
            sbContentArea.addEventListener('input', updateStatusBar);
            updateStatusBar();
        }

        // Folder modal enter to submit
        document.getElementById('folder-name-input').addEventListener('keydown', function(e) {
            if (e.key === 'Enter') {
                e.preventDefault();
                document.getElementById('folder-form').submit();
            }
        });
    </script>
